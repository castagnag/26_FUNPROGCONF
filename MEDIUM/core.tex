    \item \code{false or nil}: the values Elixir treats as false (\emph{falsy}), a union of singleton types
    \visible<3->{\item \code{not (false or nil)}: all other values (\emph{truthy}), a negation type}
\end{itemize}
\begin{block}<4->{Narrowing on the result of an application}

                     else: x + 1   # x : integer()
end
\end{minted}
\vspace{-1mm}
\end{block}

\subsection{Exhaustiveness and dead clauses}

\begin{frame}[fragile,t]
\frametitle{Guards are more than type tests}
A multi-clause function is a \code{case} on its arguments: each clause has a pattern and a
\emph{guard}.
\begin{block}{Guards combine type tests, comparisons, field access, \texttt{and}, \texttt{or}, \texttt{not}}
\begin{minted}{elixir}
def f(x) when (is_map(x) and map_size(x) == 2) or is_list(x), do: ...
\end{minted}
\end{block}
\pause
\textbf{\color{darkslateblue}The values that pass this guard do not form a type:} no type
describes ``maps of size 2''.
\pause
\begin{columns}[c,onlytextwidth]
\begin{column}{0.47\textwidth}
\centering

{\scriptsize dashed: the values that pass the guard}
\end{column}
\begin{column}{0.5\textwidth}
Two types approximate them:
\begin{itemize}
\item \surely{\textbf{surely accepted}}: \code{list()}\\
      {\small every list passes}
\item \possibly{\textbf{possibly accepted}}: \code{map() or list()}\\
      {\small only maps and lists can pass}
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[fragile,t]
\frametitle{Dead clauses and exhaustiveness}\transdissolve<2-3,6-7>
Same recipe as for patterns (slide~\matchframenumber): clause $i$ receives the input values
\textbf{minus} those \surely{surely accepted} by the previous clauses,
\textbf{and} \possibly{possibly accepted} by its own guard.
\only<1>{\begin{textblock}{11}(2.5,5.9)
\fcolorbox{gray!60}{white}{\includegraphics[width=10.8cm,trim=26 18 30 102,clip]{images/recall-match.pdf}}% overlay 5 of the [Match] frame, with the handwritten arrows (pdfseparate of main.pdf)
\end{textblock}}%
\only<2-3>{\begin{textblock}{11}(2.5,5.9)
\fcolorbox{gray!60}{white}{\scalebox{0.773}{\begin{minipage}[t][135pt]{397.5pt}% same size as the screenshot above
\vspace*{4.4pt}% aligned with the screenshot
{\setlength{\leftmargini}{\dimexpr0.75em+1.5mm}\setlength{\labelsep}{0.3em}%
\begin{itemize}\setlength{\topsep}{0pt}\setlength{\itemsep}{0.2mm}
\item To infer the type $\Bl\T_1$ of $\Bl e_1$ we need {\color{blue}$\T\And \PA{p_1}$};
\item To infer the type $\Bl\T_2$ of $\Bl e_2$ we need {\color{blue}$(\T\setminus \SA{p_1})\And \PA{p_2}$};
\item The type of the match expression is {\color{blue}$\T_1\Or \T_2$}.
\item Pattern matching is exhaustive if $\color{blue}\T\leq\PA{p_1}\Or \PA{p_2}$.
\end{itemize}}
\vspace{-4.6pt}
\hspace*{1.5mm}\textbf{Formally:}\hfill{\small $\color{typeGreen}\pmb{\lfloor} p\pmb{\rfloor}$: \surely{surely accepted}, $\color{typeOrange}\pmb{\lceil} p\pmb{\rceil}$: \possibly{possibly accepted} by $p$ and its guard}
\vspace{-2mm}
\begin{mathpar}\Bl
\infer[{[Match]}]
{\Gamma\vdash e:\T\and\Gamma, \T\!\!\And\!\!\PA{p_1\!}/p_1\vdash e_1:\T_1\and \Gamma, \T\setminus\SA{p_1\!}/p_2\vdash e_2:\T_2}
{\Gamma\vdash \texttt{match } e \texttt{ with }p_1 \texttt{->} e_1 \texttt{ | }p_2 \texttt{->} e_2 : \T_1\Or\T_2 }\mbox{\small($\T\leq\PA{p_1\!}\Or\PA{\!p_2\!}$)}
\end{mathpar}
\end{minipage}}}%
\end{textblock}}%
\only<3>{\begin{textblock}{16}(0,0)% handwritten arrows only (extracted from an annotated screenshot of overlay 2)
\includegraphics[width=\paperwidth]{images/hw-guard.png}
\end{textblock}}%
\pause\pause\pause
\begin{block}{Nothing left: a dead clause (Elixir 1.20)}
\begin{minted}[fontsize=\footnotesize]{elixir}
def red(x) when is_integer(x) or is_boolean(x), do: 1
def red(x) when is_integer(x), do: 2
\end{minted}
\vspace{-1mm}
\begin{warnverbatim}
\warn the following clause is redundant:
    def red(x) when is_integer(x)
it has type: integer()
previous clauses have already matched on the following types:
    boolean() or integer()
\end{warnverbatim}
\vspace{-1mm}
\end{block}
\pause
\vspace{-2mm}
\begin{block}{Exhaustiveness depends on the declared input type}
\begin{onlyenv}<5>
\begin{minted}[escapeinside=||,fontsize=\footnotesize]{elixir}
|\spec| list() -> _               # _ : result type left to inference
def f(x) when (is_map(x) and map_size(x) == 2) or is_list(x), do: ...
\end{minted}
\vspace{-1mm}
{\small Covered by the \surely{surely accepted} type \code{list()}: exhaustive\par}
\end{onlyenv}
\begin{onlyenv}<6>
\begin{minted}[escapeinside=||,fontsize=\footnotesize]{elixir}
|\spec| map() or list() -> _      # _ : result type left to inference
def f(x) when (is_map(x) and map_size(x) == 2) or is_list(x), do: ...
\end{minted}
\vspace{-1mm}
{\small Covered only by the \possibly{possibly accepted} type: \emph{may} not be exhaustive\par}
\end{onlyenv}
\begin{onlyenv}<7>
\begin{minted}[escapeinside=||,fontsize=\footnotesize]{elixir}
|\spec| term() -> _               # _ : result type left to inference
def f(x) when (is_map(x) and map_size(x) == 2) or is_list(x), do: ...
\end{minted}
\vspace{-1mm}
{\small Not covered: not exhaustive\par}
\end{onlyenv}
\end{block}
\end{frame}

\subsection{Narrowing and inference}

\begin{frame}[fragile,t]\transdissolve<5-6>
\frametitle{Narrowing: each clause sees a more precise type}
\vspace{-2mm}
\begin{block}{A response type}
\result
\end{block}
\begin{onlyenv}<-5>
\begin{block}{Handling a response}
\begin{minted}[escapeinside=||,fontsize=\footnotesize]{elixir}
|\spec| response() -> _
def handle(r) when r.output == :ok, do: {:accepted, r.socket}
def handle(r) when is_atom(r.message), do: r.message

\end{minted}
\vspace{-1mm}
\end{block}
\end{onlyenv}
\begin{onlyenv}<-4>
{\footnotesize\frenchspacing
\begin{tabular}{@{}c@{\quad}l@{\quad}l@{}}
clause & type of \texttt{r} & result\\\hline
\uncover<2->{1 & \texttt{\%\{output: :ok, socket: socket()\}} & \texttt{\{:accepted, socket()\}}}\\
\uncover<3->{2 & \texttt{\%\{output: :error, message: :timeout\}} & \texttt{:timeout}}\\
\uncover<4->{3 & \texttt{\%\{output: :error, message: \{:delay, integer()\}\}} & \texttt{\{:retry, integer()\}}}

\end{onlyenv}
\begin{onlyenv}<5>

\end{onlyenv}

\begin{frame}[fragile,t]
\frametitle{Splitting guards: several arrows for one clause}
\begin{block}{One clause, a disjunctive guard}
\begin{minted}{elixir}
def baz(x, y) when is_boolean(x) or is_integer(y), do: {y, x}
\end{minted}
\end{block}
\pause
The guard is split into its \texttt{or} branches; the body is typed once per branch:

\medskip
{\small
\begin{tabular}{@{}l@{\quad}l@{}}
\textcolor{typeBlue}{\code{is_boolean(x)}} & \textcolor{typeBlue}{\texttt{(boolean(), term() -> \{term(), boolean()\})}}\\[1mm]
\uncover<3->{\textcolor{bordeaux}{\code{is_integer(y)}}} & \uncover<3->{\textcolor{bordeaux}{\texttt{(term(), integer() -> \{integer(), term()\})}}}
\end{tabular}\par}
\pause\pause
\begin{onlyenv}<-4>
\begin{exampleblock}{Inferred type: the intersection of the two arrows}
\begin{coloredverbatim}
$ (boolean(), term() -> \{term(), boolean()\}) \oand
  (term(), integer() -> \{integer(), term()\})
\end{coloredverbatim}
\vspace{-1mm}
\end{exampleblock}
{\small\color{gray}Branches are analyzed in evaluation order: swapping the operands of
\texttt{or} can change the arrows.\par}
\end{onlyenv}
\begin{onlyenv}<5>
\begin{block}{What it allows to type}
\begin{minted}[fontsize=\footnotesize]{elixir}
{n, b} = baz(true, 2)   # both arrows apply: {integer(), boolean()}
{n + 1, not b}          # accepted (with one arrow: n, b : term(), rejected)
\end{minted}
\vspace{-1mm}
\end{block}
\end{onlyenv}
\end{frame}
